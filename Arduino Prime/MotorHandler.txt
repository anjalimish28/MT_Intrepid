module Intrepid {
    
    @ ESCON motor controller driver component
    @ Interfaces with motor controllers via PWM, GPIO, and I2C ADC sensors
    passive component MotorHandler {

        ###############################################################################
        # Input Ports                                                                 #
        ###############################################################################

        @ Motor command input port - receives commands for all 3 motors
        sync input port motor_cmd: RDC.MotorCommand

        @ Rate group input port
        sync input port run: Svc.Sched

        ###############################################################################
        # Output Ports - Motor Control                                                #
        ###############################################################################

        @ PWM duty cycle output for MBS motor speed
        output port pwm_mbs: Drv.DutyCycle

        @ PWM duty cycle output for Gimbal motor speed
        output port pwm_gimbal: Drv.DutyCycle

        @ PWM duty cycle output for Flywheel motor speed
        output port pwm_flywheel: Drv.DutyCycle

        ###############################################################################
        # Output Ports - GPIO Control                                                 #
        ###############################################################################

        @ GPIO output for MBS motor enable
        output port gpio_mbs_enable: Drv.GpioWrite

        @ GPIO output for MBS motor direction
        output port gpio_mbs_direction: Drv.GpioWrite

        @ GPIO output for Gimbal motor enable
        output port gpio_gimbal_enable: Drv.GpioWrite

        @ GPIO output for Gimbal motor direction
        output port gpio_gimbal_direction: Drv.GpioWrite

        @ GPIO output for Flywheel motor enable
        output port gpio_flywheel_enable: Drv.GpioWrite

        @ GPIO output for Flywheel motor direction
        output port gpio_flywheel_direction: Drv.GpioWrite

        ###############################################################################
        # Output Ports - Telemetry                                                    #
        ###############################################################################

        @ Speed data output port for external consumers (sent during run)
        output port speed_data: RDC.SpeedData

        ###############################################################################
        # Test Commands                                                               #
        ###############################################################################

        sync command TEST_PWM(
            motor_id:   RDC.MotorSelect,            @< Motor ID (0=MBS, 1=Gimbal, 2=Flywheel)
            enable:     Fw.On,                      @< Enable
            direction:  RDC.MotorDirection,         @< Direction
            set_speed:  U16                         @< Motor RPM setpoint (maps to 10-90% PWM duty cycle)
        ) 

        ###############################################################################
        # Telemetry Channels                                                          #
        ###############################################################################

        telemetry ESCON_MBS_CURRENT_SPEED: F32
        telemetry ESCON_GIMBAL_CURRENT_SPEED: F32
        telemetry ESCON_GIMBAL_AVERAGE_SPEED: F32
        telemetry ESCON_GIMBAL_ROTATION: F32
        telemetry ESCON_FLYWHEEL_CURRENT_SPEED: F32
        telemetry ESCON_FLYWHEEL_AVERAGE_SPEED: F32

        ###############################################################################
        # Standard AC Ports: Required for Channels, Events, Commands, and Parameters  #
        ###############################################################################

        @ Port for requesting the current time
        time get port timeCaller

        @ Ports for Commanding
        import Fw.Command

        @ Ports for events
        import Fw.Event

        @ Ports for telemetry
        import Fw.Channel

    }
}
