module CanNode {

  # ----------------------------------------------------------------------
  # Base ID Convention
  # ----------------------------------------------------------------------
  #
  # All Base IDs follow the 8-digit hex format: 0xDSSCCxxx
  #
  # Where:
  #   D   = Deployment digit (1 for this deployment)
  #   SS  = Subtopology digits (00 for main topology, 01-05 for subtopologies)
  #   CC  = Component digits (00, 01, 02, etc.)
  #   xxx = Reserved for internal component items (events, commands, telemetry)
  #
  #   100XXXXX = Application
  #   101XXXXX = Cmd_Core components
  #   102XXXXX = Scheduling components
  #   103XXXXX = Interface Driver components
  #   104XXXXX = Hub pattern

  # ----------------------------------------------------------------------
  # Defaults
  # ----------------------------------------------------------------------

  module Default {
    constant QUEUE_SIZE = 3
    constant STACK_SIZE = 64 * 1024
  }

  # ----------------------------------------------------------------------
  # Active component instances
  # ----------------------------------------------------------------------

  # Application: Test Led
  instance led: Components.Led base id 0x10001000 \
    queue size Default.QUEUE_SIZE \
    stack size Default.STACK_SIZE \
    priority 95

  # Cmd_Core
  instance cmdDisp: Svc.CommandDispatcher base id 0x10100000 \
    queue size Default.QUEUE_SIZE\
    stack size Default.STACK_SIZE \
    priority 101

  instance eventLogger: Svc.EventManager base id 0x10101000 \
    queue size Default.QUEUE_SIZE \
    stack size Default.STACK_SIZE \
    priority 98

  instance tlmSend: Svc.TlmChan base id 0x10102000 \
    queue size Default.QUEUE_SIZE \
    stack size Default.STACK_SIZE \
    priority 97

  # ----------------------------------------------------------------------
  # Queued component instances
  # ----------------------------------------------------------------------

  # ----------------------------------------------------------------------
  # Passive component instances
  # ----------------------------------------------------------------------

  # Application
  instance axisHandler: ArduinoPrime.InterfaceTester base id 0x10000000

  # Cmd_Core
  instance fatalHandler: Baremetal.FatalHandler base id 0x10103000

  instance textLogger: Svc.PassiveTextLogger base id 0x10104000

  # Scheduling
  instance timeHandler: Arduino.ArduinoTime base id 0x10200000

  instance systemResources: Svc.SystemResources base id 0x10201000

  instance rateDriver: Arduino.HardwareRateDriver base id 0x10202000

  instance rateGroupDriver: Svc.RateGroupDriver base id 0x10203000

  instance rateGroup1: Svc.PassiveRateGroup base id 0x10204000

  # Interface Driver
  instance comDriver: Arduino.StreamDriver base id 0x10300000

  instance canDriver: Arduino.CanDriver base id 0x10301000

  instance pwmDriver1: Arduino.PwmDriver base id 0x10302000

  instance pwmDriver2: Arduino.PwmDriver base id 0x10303000

  instance pwmDriver3: Arduino.PwmDriver base id 0x10304000

  instance gpioDriver1: Arduino.GpioDriver base id 0x10305000

  instance gpioDriver2: Arduino.GpioDriver base id 0x10306000

  instance gpioDriver3: Arduino.GpioDriver base id 0x10307000

  instance gpioDriver4: Arduino.GpioDriver base id 0x10308000

  instance gpioLedDriver5: Arduino.GpioDriver base id 0x10309000

  instance gpioDriver6: Arduino.GpioDriver base id 0x10310000

  # Hub Pattern
  instance hubBufferManager: Svc.BufferManager base id 0x10400000 \
  {
      phase Fpp.ToCpp.Phases.configObjects """
      Svc::BufferManager::BufferBins bins;
      """
      
      phase Fpp.ToCpp.Phases.configComponents """
      memset(&ConfigObjects::CanNode_hubBufferManager::bins, 0, sizeof(ConfigObjects::CanNode_hubBufferManager::bins));
      ConfigObjects::CanNode_hubBufferManager::bins.bins[0].bufferSize = CandoSvc::CanMailboxCfg::CanMailboxSmallBufferSize;
      ConfigObjects::CanNode_hubBufferManager::bins.bins[0].numBuffers = CandoSvc::CanMailboxCfg::CanMailboxSmallBufferCount;
      ConfigObjects::CanNode_hubBufferManager::bins.bins[1].bufferSize = CandoSvc::CanMailboxCfg::CanMailboxMediumBufferSize;
      ConfigObjects::CanNode_hubBufferManager::bins.bins[1].numBuffers = CandoSvc::CanMailboxCfg::CanMailboxMediumBufferCount;
      ConfigObjects::CanNode_hubBufferManager::bins.bins[2].bufferSize = CandoSvc::CanMailboxCfg::CanMailboxBigBufferSize;
      ConfigObjects::CanNode_hubBufferManager::bins.bins[2].numBuffers = CandoSvc::CanMailboxCfg::CanMailboxBigBufferCount;
      CanNode::hubBufferManager.setup(
          42,
          6,
          CanNode::Allocation::memAllocator,
          ConfigObjects::CanNode_hubBufferManager::bins
      );
      """

      phase Fpp.ToCpp.Phases.tearDownComponents """
      CanNode::hubBufferManager.cleanup();
      """
  }

  instance canMailbox: CandoSvc.CanMailbox base id 0x10401000

  instance hubRdc3: Svc.GenericHub base id 0x10402000

}
