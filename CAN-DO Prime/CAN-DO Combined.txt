module Drv {

    @ CAN header struct
    struct CanHeader {
        isExtended:     bool    @< Extended id flag (true for extended)
        isRemoteFrame:  bool    @< Remote flag (true for remote frame)
        $id:            U32     @< CAN message id (11 bit standard; 29 bit extended)
        dataLength:     U8      @< CAN payload length (O to 8 bytes possible)
        timestamp:      U16     @< Receive timestamp (not used)
        device:         U8      @< CAN Controller id (dependent on peripherals)
    } default {isExtended = false, isRemoteFrame = false, $id = 0, dataLength = 0, timestamp = 0, device = 0}

    @ CAN Payload Array
    array CanPayload = [8] U8 default 0 format "0x{} "

    @ This port is used for exchanging CAN messages with driver
    port CanMessage(
        ref header: CanHeader
        ref data:   CanPayload
    ) -> ByteStreamStatus

    port CanFilter(
        bankNum:        U8      @< Bank/Mailbox number
        $id:            U32     @< Partial or full CAN id to match (11 or 29 bit)
        mask:           U32     @< Filter mask fitting the id match pattern (11 or 29 bit)
        isGlobal:       bool    @< Global filter flag
        isExtended:     bool    @< Extended flag
        $priority:      U8      @< Priority number (implementation dependent on CAN peripheral)
        enable:         bool    @< Enable flag (false to disable filter on mailbox)
    ) -> bool                   @< Return success flag

    @ Synchronous CAN input interface
    interface CanDriver {

        @ Set CAN rx mailbox filter
        @ Success flag is returned
        guarded input port setRxFilter: CanFilter

        @ Port to invoke readout of CAN buffer by reference
        @ Status is returned
        guarded input port canRx: CanMessage

        @ Invoke this port to send data out the driver (synchronous)
        @ Status is returned
        guarded input port canTx: CanMessage

        @ Port invoked when the driver is ready to send/receive data
        output port ready: Drv.ByteStreamReady

    }

    @ Synchronous CAN output interface
    interface CanManager {

        @ Ready signal when CAN driver is connected
        sync input port drvConnected: Drv.ByteStreamReady
        
        @ Set CAN mailbox filter
        output port setRxFilter: CanFilter

        @ Port to invoke CAN receive
        output port canRx: CanMessage

        @ Port to invoke CAN transmit
        output port canTx: CanMessage
        
    }
}

module CandoSvc {

  @| CAN Mailbox component
  @| ----------------------------------------------------------------------
  @| The Can Mailbox component provides an extension of the GenericHub pattern for CAN communication.
  @| Instead of fixed pairs of hubs on different deployments, the CAN Mailbox provides outgoing and
  @| incoming mailboxes that represent destination and source addresses of deployment nodes.
  @| This way the processor and hubs are relieved as the CAN hardware will filter the relevant messages
  @| and only trigger the mailbox when messages of interest are received.
  @| This extended pattern will require an GenericHub per target node and for multicasting.
  @|
    passive component CanMailbox {

        @ This interface provides ports allocate and deallocate
        import Svc.BufferAllocation

        @ CAN interface
        import Drv.CanManager

        ###############################################################################
        # Input Ports                                                                 #
        ###############################################################################

        @ Polling mailbox for new messages
        sync input port pollMb: Svc.Sched
        
        @ Port receiving back ownership of data buffer sent out on incomingMb port
        guarded input port recvReturnIn: [CanMailboxCfg.NumOfIncomingMailboxes] Fw.BufferSend

        @ Invoke this port to send data out on mailbox (synchronous)
        guarded input port outgoingMb: [CanMailboxCfg.NumOfOutgoingMailboxes] Fw.BufferSend

        ###############################################################################
        # Output Ports - Telemetry                                                    #
        ###############################################################################

        @ Port invoked by the mailbox when it receives data
        output port incomingMb: [CanMailboxCfg.NumOfIncomingMailboxes] Fw.BufferSend

        @ Port to return ownership of data buffer received from outgoingMb port
        output port sendReturnOut: [CanMailboxCfg.NumOfOutgoingMailboxes] Fw.BufferSend

        match incomingMb with recvReturnIn

        match sendReturnOut with outgoingMb

        ###############################################################################
        # Events                                                                      #
        ###############################################################################

        @ CAN receive debug
        event CanReceiveDebug(
                            device:     U8              @< CAN controller number
                            $id:        U32             @< CAN message ID
                            timestamp:  U16             @< Timestamp of the CAN message
                            dataLength: U8              @< Length of the CAN message data
                            data:       Drv.CanPayload  @< The data buffer of the CAN message
                            ) \
        severity diagnostic \
        id 1 \
        format "CAN recv ctrl {}; id {}, timestamp {}, len {}, data {}"

        @ CAN received non configured message
        event CanReceiveNoMb(
                            device:     U8              @< CAN controller number
                            $id:        U32             @< CAN message ID
                            timestamp:  U16             @< Timestamp of the CAN message
                            dataLength: U8              @< Length of the CAN message data
                            data:       Drv.CanPayload  @< The data buffer of the CAN message
                            ) \
        severity warning high \
        id 2 \
        format "CAN no MB ctrl {}; id {}, timestamp {}, len {}, data {}"

        @ CAN outgoing message failure
        event CanOutgoingFailure(mbNum: U32) severity warning high format "CAN mailbox {} outgoing message failure!"

        ###############################################################################
        # Standard AC Ports: Required for Channels, Events, Commands, and Parameters  #
        ###############################################################################

        @ Port for requesting the current time
        time get port timeCaller

        @ Ports for Commanding
        import Fw.Command

        @ Ports for events
        import Fw.Event

        @ Ports for telemetry
        import Fw.Channel

    }
}

module CandoSvc {

  module CanMailboxCfg {

    # Less than 256 to fit in U8
    constant NumOfOutgoingMailboxes = 14
    constant NumOfIncomingMailboxes = 14

    constant CanMailboxSmallBufferSize = 32
    constant CanMailboxSmallBufferCount = 10
    constant CanMailboxMediumBufferSize = 128
    constant CanMailboxMediumBufferCount = 5
    constant CanMailboxBigBufferSize = 512
    constant CanMailboxBigBufferCount = 3

  }

}

module RDC {

    @ Motor selection enum
    enum MotorSelect : U8 {
        MBS         @< MBS
        GIMBAL      @< Gimbal
        FLYWHEEL    @< Flywheel
    }

    @ Motor direction state
    enum MotorDirection : U8 {
        REVERSE @< Reverse state
        FORWARD @< Forward state
    }

    @ Custom port for motor command interface
    port MotorCommand(
        motor_id:   MotorSelect,        @< Motor ID (0=MBS, 1=Gimbal, 2=Flywheel)
        enable:     Fw.On,              @< Enable flag (0=disabled, 1=enabled)
        direction:  MotorDirection,     @< Direction (0=reverse, 1=forward)
        set_speed:  U16                 @< Motor RPM setpoint (maps to 10-90% PWM duty cycle)
    )

    @ Custom port for motor speed telemetry data output
    port SpeedData(
        motor_id: U8,           @< Motor ID (0=MBS, 1=Gimbal, 2=Flywheel)
        current_speed: F32,     @< Current actual speed reading (RPM)
        average_speed: F32,     @< Average speed reading (RPM) (only Gimbal and Flywheel)
        encoder_value: F32      @< Encoder/rotation sensor value (only Gimbal)
    )

}
