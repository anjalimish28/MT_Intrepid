module Va416x0Drv {

    @ CAN Driver for Va416x0.
    passive component CanDriver {

        @ Ports for CAN driver interface
        import Drv.CanDriver

        @ Runtime initialization
        sync input port initialize : Fw.Ready

        @ Port invoked by CAN interrupt when a frame was received
        sync input port canIrq: Va416x0Svc.ExceptionHandler

        @ CAN controller error
        event CanError(
                        controller:     U8      @< CAN controller number
                        rxErrCnt:       U8      @< CAN rx error count
                        txErrCnt:       U8      @< CAN tx error count
                        errField:       U16     @< Error diagnostic register
                                ) \
        severity warning high \
        id 0 \
        format "CAN error at controller {}; RX error count {}; TX error count {}; error diagnostic 0x{x}"

        @ CAN message test
        event CanTestMsg(
                        device:         U8              @< CAN controller number
                        $id:            U32             @< CAN message ID
                        timestamp:      U16             @< Timestamp of the CAN message
                        dataLength:     U8              @< Length of the CAN message data
                        data:           Drv.CanPayload  @< The data buffer of the CAN message
                            ) \
        severity activity low \
        id 1 \
        format "CAN test msg ctrl {}; id {}, timestamp {}, len {}, data {}"

        ###############################################################################
        # Standard AC Ports: Required for Channels, Events, Commands, and Parameters  #
        ###############################################################################
        
        @ Port for requesting the current time
        time get port timeCaller

        @ Ports for events
        import Fw.Event

        @ Ports for telemetry
        import Fw.Channel

    }
}
