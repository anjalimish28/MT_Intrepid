    @ CAN Driver for Va416x0.
    passive component CanDriver {

        @ Ports for CAN driver interface
        import Drv.CanDriver

        @ Runtime initialization
        sync input port initialize : Fw.Ready

        @ Port invoked by CAN interrupt when a frame was received
        sync input port canIrq: Va416x0Svc.ExceptionHandler

        @ CAN controller error
        event CanError(
                        controller:     U8      @< CAN controller number
                        rxErrCnt:       U8      @< CAN rx error count
                        txErrCnt:       U8      @< CAN tx error count
                        errField:       U16     @< Error diagnostic register
                                ) \
        severity warning high \
        id 0 \
        format "CAN error at controller {}; RX error count {}; TX error count {}; error diagnostic 0x{x}"

        @ CAN message test
        event CanTestMsg(
                        device:         U8              @< CAN controller number
                        $id:            U32             @< CAN message ID
                        timestamp:      U16             @< Timestamp of the CAN message
                        dataLength:     U8              @< Length of the CAN message data
                        data:           Drv.CanPayload  @< The data buffer of the CAN message
                            ) \
        severity activity low \
        id 1 \
        format "CAN test msg ctrl {}; id {}, timestamp {}, len {}, data {}"

        ###############################################################################
        # Standard AC Ports: Required for Channels, Events, Commands, and Parameters  #
        ###############################################################################
        
        @ Port for requesting the current time
        time get port timeCaller

        @ Ports for events
        import Fw.Event

        @ Ports for telemetry
        import Fw.Channel

    }

  # Base ID Convention
  # ----------------------------------------------------------------------
  #
  # All Base IDs follow the 8-digit hex format: 0xDSSCCxxx
  #
  # Where:
  #   D   = Deployment digit (1 for this deployment)
  #   SS  = Subtopology digits (00 for main topology, 01-05 for subtopologies)
  #   CC  = Component digits (00, 01, 02, etc.)
  #   xxx = Reserved for internal component items (events, commands, telemetry)
  #
  #   100XXXXX = Application
  #   101XXXXX = Cmd_Core components
  #   102XXXXX = Scheduling components
  #   103XXXXX = Interface Driver components
  #   104XXXXX = Hub Pattern
  #   105XXXXX = Exception/Interrupt Handling components

module Va416x0Drv {
    @ Uart driver for VA416x0.
    passive component UartDriver {

        @ Polling for receiving data
        sync input port schedIn: Svc.Sched

        @ Port invoked by dma interrupt when a receive cycle is complete
        sync input port dmaDoneIrq: Va416x0Svc.ExceptionHandler

        @ Runtime initialization
        sync input port initialize : Fw.Ready

        import Drv.ByteStreamDriver

        @ Allocation port used for allocating memory in the receive task
        output port allocate: Fw.BufferGet

        @ Deallocation of allocated buffers
        output port deallocate: Fw.BufferSend

        @ Start DMA Channel for RX
        output port startRxDma: Va416x0Drv.StartDmaTransaction

        @ Status DMA Channel for RX
        output port statusRxDma: Va416x0Drv.StatusDmaTransaction

        @ Stop DMA Channel for RX
        output port stopRxDma: Va416x0Drv.StopDmaTransaction

        telemetry port Tlm
        time get port Time

        telemetry BytesReceived: U32 update on change
        telemetry FailedReceives: U32 update on change
        telemetry BytesSent: U32 update on change
        telemetry FailedSends: U32 update on change

    }
}

module Va416x0Svc {

    @ Port allowing a component to set the time
    port SetTime(
        year:  U32, @< The current year
        month: U8,  @< The current month (1-12)
        day: U8,    @< The current day (1-31)
        hour: U8,   @< The current hour (0-23)
        minute: U8, @< The current minute (0-59)
        second: U8, @< The current second (0-59)
    )

    @ A component for getting time of Va416x0.
    passive component Va416x0Time {

        import Svc.Time

        @ Port to set the time
        sync input port setTime: SetTime

        @ Command to set the time
        sync command SET_TIME(
                year:  U32, @< The current year
                month: U8,  @< The current month (1-12)
                day: U8,    @< The current day (1-31)
                hour: U8,   @< The current hour (0-23)
                minute: U8, @< The current minute (0-59)
                second: U8, @< The current second (0-59)
        )

        @ Command to set the time
        event TimeUpdate(
                original_seconds:      U32  @< The seconds before the time update
                original_microseconds: U32  @< The microseconds before the time update
                original_base:          U8  @< The time base before the update
                new_seconds:           U32  @< The seconds after the time update
                new_microseconds:      U32  @< The microseconds after the time update
                new_base:               U8  @< The time base after the update
        ) severity activity high format "Time updated from {}.{} ({}) to {}.{} ({})"

        ###############################################################################
        # Standard AC Ports: Required for Channels, Events, Commands, and Parameters  #
        ###############################################################################
        @ Port for requesting the current time
        time get port timeCaller

        @ Enables command handling
        import Fw.Command

        @ Enables event handling
        import Fw.Event

        @ Enables telemetry channels handling
        import Fw.Channel

    }
}
